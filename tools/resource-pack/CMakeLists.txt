cmake_minimum_required(VERSION 3.20)
project(XLResourceEncoder LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
set(ZSTD_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(ZSTD_BUILD_DICTBUILDER OFF CACHE BOOL "" FORCE)
add_subdirectory("${XL_ZSTD_SOURCE}/build/cmake" zstd)
add_executable(xl_resource_pack ../resource_pack.cpp)
target_link_libraries(xl_resource_pack PRIVATE libzstd_static)
if(MSVC)
  target_compile_options(xl_resource_pack PRIVATE /utf-8)
  target_link_options(xl_resource_pack PRIVATE /Brepro)
endif()

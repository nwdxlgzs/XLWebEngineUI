# Static archives ignore LINK_DEPENDS. Recompile an empty generated source
# whenever an input archive or the merger changes, so POST_BUILD runs again.
set(XL_PRIVATE_INPUT_STAMP "${CMAKE_CURRENT_BINARY_DIR}/private_sdk_inputs.cpp")
add_custom_command(OUTPUT "${XL_PRIVATE_INPUT_STAMP}"
  COMMAND "${CMAKE_COMMAND}" -E touch "${XL_PRIVATE_INPUT_STAMP}"
  DEPENDS xl_zstd_decoder xl_skia_pathops "$<TARGET_FILE:xl_skia_pathops>" ${XL_ICU_LIBRARIES} ${XL_GRAPHICS_LIBRARIES} "${PROJECT_SOURCE_DIR}/tools/merge_archive.py"
  VERBATIM)
target_sources(xl_static PRIVATE "${XL_PRIVATE_INPUT_STAMP}")
add_custom_command(TARGET xl_static POST_BUILD
  COMMAND "${Python3_EXECUTABLE}" -B "${PROJECT_SOURCE_DIR}/tools/merge_archive.py"
    --output "$<TARGET_FILE:xl_static>" --libraries "$<TARGET_FILE:xl_zstd_decoder>" "$<TARGET_FILE:xl_skia_pathops>" ${XL_ICU_LIBRARIES} ${XL_GRAPHICS_LIBRARIES}
    --isolate "$<TARGET_FILE:xl_zstd_decoder>" "$<TARGET_FILE:xl_skia_pathops>" ${XL_ICU_LIBRARIES} ${XL_GRAPHICS_LIBRARIES}
    --ar "${XL_MERGE_AR}" --nm "${XL_MERGE_NM}" --objcopy "${XL_MERGE_OBJCOPY}"
  VERBATIM COMMAND_EXPAND_LISTS)

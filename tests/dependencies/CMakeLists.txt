cmake_minimum_required(VERSION 3.20)
project(XLDependencyIntegration LANGUAGES C CXX)
include(CTest)
include("${XL_DEPENDENCY_MANIFEST}")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_package(SDL3 CONFIG REQUIRED PATHS "${XL_SDL_PREFIX}" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
find_package(Threads REQUIRED)
add_library(xl_dependency_set INTERFACE)
target_compile_definitions(xl_dependency_set INTERFACE U_STATIC_IMPLEMENTATION U_LIB_SUFFIX_C_NAME=_xl SDL_MAIN_HANDLED LEXBOR_STATIC)
target_include_directories(xl_dependency_set INTERFACE "${XL_SKIA_ROOT}" ${XL_DEP_INCLUDE_DIRS})
if(MSVC)
  target_compile_options(xl_dependency_set INTERFACE /utf-8)
  target_link_libraries(xl_dependency_set INTERFACE ${XL_DEP_LIBRARIES} SDL3::SDL3-static
    dwrite usp10 ole32 oleaut32 user32 gdi32 opengl32 fontsub uuid)
else()
  target_link_libraries(xl_dependency_set INTERFACE "-Wl,--start-group" ${XL_DEP_LIBRARIES}
    SDL3::SDL3-static ${XL_PLATFORM_LIBRARIES} "-Wl,--end-group" Threads::Threads m dl)
endif()
add_executable(xl_dependency_integration main.cpp)
target_link_libraries(xl_dependency_integration PRIVATE xl_dependency_set)
add_test(NAME dependencies.real_calls COMMAND xl_dependency_integration)
set_tests_properties(dependencies.real_calls PROPERTIES TIMEOUT 60 LABELS "dependencies;graphics;text;script;media")
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../third_party/recipes/zstd-runtime" zstd-runtime)
add_executable(xl_resource_service_tests ../resources/service_tests.cpp
  ../../src/resources/service.cpp ../../src/base/executor.cpp ../../src/base/memory_pool.cpp)
target_include_directories(xl_resource_service_tests PRIVATE ../../src)
target_link_libraries(xl_resource_service_tests PRIVATE xl_zstd_decoder xl_dependency_set)
if(MSVC)
  target_compile_options(xl_resource_service_tests PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(xl_resource_service_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME resources.lazy_service COMMAND xl_resource_service_tests)
set_tests_properties(resources.lazy_service PROPERTIES TIMEOUT 45 LABELS "resources;concurrency;lifecycle")
add_executable(xl_resource_pack ../../tools/resource_pack.cpp)
target_link_libraries(xl_resource_pack PRIVATE xl_dependency_set)
if(XL_EMBEDDED_CATALOG)
  add_executable(xl_builtin_resources ../resources/builtin_tests.cpp "${XL_EMBEDDED_CATALOG}/catalog.cpp"
    ../../src/resources/service.cpp ../../src/base/executor.cpp ../../src/base/memory_pool.cpp)
  target_include_directories(xl_builtin_resources PRIVATE ../../src "${XL_EMBEDDED_CATALOG}")
  target_link_libraries(xl_builtin_resources PRIVATE xl_zstd_decoder xl_dependency_set)
  add_test(NAME resources.embedded_icu COMMAND xl_builtin_resources)
  set_tests_properties(resources.embedded_icu PROPERTIES TIMEOUT 45 LABELS "resources;text;lifecycle")
endif()
option(XL_TEST_NATIVE_WINDOW "Require real native presentation and child embedding" OFF)
if(XL_TEST_NATIVE_WINDOW)
  add_executable(xl_native_window native_window.cpp)
  target_link_libraries(xl_native_window PRIVATE xl_dependency_set)
  add_test(NAME dependencies.native_window COMMAND xl_native_window)
  set_tests_properties(dependencies.native_window PROPERTIES TIMEOUT 30 LABELS "dependencies;graphics;native;embedding")
endif()

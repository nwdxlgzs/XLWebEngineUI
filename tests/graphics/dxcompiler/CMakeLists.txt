cmake_minimum_required(VERSION 3.21)
cmake_policy(SET CMP0091 NEW)
project(XLStaticDXCProbe LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_package(Threads REQUIRED)
add_executable(xl_dxcompiler_static ../dxcompiler_static.cpp)
target_compile_features(xl_dxcompiler_static PRIVATE cxx_std_17)
target_include_directories(xl_dxcompiler_static PRIVATE "${XL_DXC_PREFIX}/include")
target_link_libraries(xl_dxcompiler_static PRIVATE "${XL_DXC_LIBRARY}" Threads::Threads)
if(WIN32)
  target_link_libraries(xl_dxcompiler_static PRIVATE ole32 oleaut32 uuid version advapi32 user32 psapi dbghelp)
else()
  target_link_libraries(xl_dxcompiler_static PRIVATE dl)
endif()
enable_testing()
add_test(NAME graphics.dxcompiler_static COMMAND xl_dxcompiler_static)
set_tests_properties(graphics.dxcompiler_static PROPERTIES TIMEOUT 60)

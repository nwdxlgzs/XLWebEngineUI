cmake_minimum_required(VERSION 3.20)
project(XLSDKConsumer LANGUAGES C CXX)
find_package(XLWebEngineUI CONFIG REQUIRED)
enable_testing()
file(TO_CMAKE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/test-font.ttf" XL_TEST_FONT)
if(NOT EXISTS "${XL_TEST_FONT}")
  message(FATAL_ERROR "Consumer test font must be copied from the locked Skia input")
endif()
foreach(consumer IN ITEMS consume_static consume_shared consume_static_c consume_shared_cpp)
  if(consumer STREQUAL "consume_static" OR consumer STREQUAL "consume_shared_cpp")
    set(source main.cpp)
  else()
    set(source main.c)
  endif()
  add_executable(${consumer} ${source} host_symbols.c host_skia.cpp)
  target_compile_definitions(${consumer} PRIVATE "XL_TEST_FONT=\"${XL_TEST_FONT}\"")
  if(consumer MATCHES "^consume_static")
    target_link_libraries(${consumer} PRIVATE XLWebEngineUI::Static)
  else()
    target_link_libraries(${consumer} PRIVATE XLWebEngineUI::Shared)
  endif()
  if(MSVC)
    set_property(TARGET ${consumer} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
  add_test(NAME sdk.${consumer} COMMAND ${consumer})
endforeach()
